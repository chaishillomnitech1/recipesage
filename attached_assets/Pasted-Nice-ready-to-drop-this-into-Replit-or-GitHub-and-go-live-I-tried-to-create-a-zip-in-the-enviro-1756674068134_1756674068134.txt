File: contracts/IReceipt.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;
import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract IReceipt is ERC721, Ownable {
    uint256 public nextId;
    mapping(uint256 => bytes32) public ceremonyHash;

    event ReceiptMinted(address indexed to, uint256 indexed tokenId, string uri, bytes32 ceremonyHash);

    constructor() ERC721("ScrollVerse Receipt","IREC") {
        nextId = 1;
    }

    function mintReceipt(address to, string calldata uri, bytes32 _ceremonyHash) external onlyOwner returns (uint256) {
        uint256 id = nextId;
        nextId++;
        ceremonyHash[id] = _ceremonyHash;
        _safeMint(to, id);
        emit ReceiptMinted(to, id, uri, _ceremonyHash);
        return id;
    }
}

import { useState } from 'react';
import { ethers } from 'ethers';

export default function MintButton(){
  const [loading,setLoading] = useState(false);
  const handleMint = async () => {
    try {
      setLoading(true);
      if(!window.ethereum) throw new Error("No wallet found");
      await window.ethereum.request({method:'eth_requestAccounts'});
      // call backend /api/ignite which pins to IPFS and mints via PRIVATE_KEY owner
      const res = await fetch('/api/ignite', {method:'POST'});
      const json = await res.json();
      alert('Ignite response: ' + JSON.stringify(json));
    } catch(e){
      alert(e.message || e);
    } finally {
      setLoading(false);
    }
  }
  return <button onClick={handleMint} style={{padding:'12px 20px', fontSize:16}} disabled={loading}>{loading ? 'Working...' : 'Mint Virgoat Genesis (Token #1)'}</button>
}

File: pages/api/ignite.js

NOTE: This file contains placeholders for IPFS pinning and expects your PRIVATE_KEY & CONTRACT_ADDRESS in env. Replace pinning call with Pinata/Infura as needed.

// /api/ignite
import { ethers } from 'ethers';
import fs from 'fs';
import path from 'path';

export default async function handler(req, res){
  if(req.method !== 'POST') return res.status(405).end();
  try{
    // 1) Load ceremony data from a "DB" (placeholder JSON in repo)
    const dbPath = path.join(process.cwd(),'data','ceremony_sample.json');
    let ceremony = {title:'Virgoat Genesis', ownerAddress: process.env.OWNER_ADDRESS || '', timestamp: Date.now()};
    if(fs.existsSync(dbPath)){
      ceremony = JSON.parse(fs.readFileSync(dbPath,'utf8'));
    }

    // 2) Compute ceremony hash
    const ceremonyJson = JSON.stringify(ceremony);
    const ceremonyBytes = ethers.utils.toUtf8Bytes(ceremonyJson);
    const ceremonyHash = ethers.utils.keccak256(ceremonyBytes);

    // 3) IPFS pin placeholder - in production replace with Pinata/Infura API call
    const ipfsPlaceholder = 'ipfs://QmPLACEHOLDER_' + Date.now();

    // 4) On-chain mint via CONTRACT_ADDRESS (owner calls mintReceipt)
    const provider = new ethers.providers.JsonRpcProvider(process.env.RPC_URL || 'http://127.0.0.1:8545');
    const wallet = new ethers.Wallet(process.env.PRIVATE_KEY || '0x' + '11'.repeat(32), provider);
    const abi = JSON.parse(fs.readFileSync(path.join(process.cwd(),'IReceiptABI.json'),'utf8'));
    const contractAddress = process.env.CONTRACT_ADDRESS || '';
    if(!contractAddress) {
      return res.status(500).json({error:'CONTRACT_ADDRESS not set in env'});
    }
    const contract = new ethers.Contract(contractAddress, abi, wallet);

    const tx = await contract.mintReceipt(ceremony.ownerAddress || wallet.address, ipfsPlaceholder, ceremonyHash);
    await tx.wait();

    return res.json({ok:true, txHash: tx.hash, ipfs: ipfsPlaceholder, ceremonyHash});
  }catch(e){
    console.error(e);
    return res.status(500).json({error:e.message});
  }
}

File: pages/api/verify.js

import fs from 'fs';
import path from 'path';
import { ethers } from 'ethers';

export default async function handler(req,res){
  if(req.method !== 'POST') return res.status(405).end();
  const body = req.body ? (typeof req.body === 'object' ? req.body : JSON.parse(req.body)) : {};
  const { id, fields } = body;
  const dbPath = path.join(process.cwd(),'data','ceremony_sample.json');
  if(!fs.existsSync(dbPath)) return res.json({match:false, reason:'no ceremony on file'});
  const ceremony = JSON.parse(fs.readFileSync(dbPath,'utf8'));
  const computed = ethers.utils.keccak256(ethers.utils.toUtf8Bytes(JSON.stringify(fields)));
  const storedHash = ceremony.ceremonyHash || null;
  return res.json({match: storedHash === computed, computed, storedHash});
}

# ScrollVerse Starter (Replit-ready)

This starter repo includes:
- Next.js frontend with `/api/ignite` to pin + mint (placeholder)
- IReceipt.sol ERC721 receipt contract + Hardhat deploy script
- Engraving/NFC template and crest SVG
- Example admin UI and mint button

## Quickstart (Replit)
1. Create a new Replit (Import from GitHub or paste files)
2. Add environment variables from `.env.example` via Replit secrets.
3. `Run` the project (Next dev server will start).
4. To deploy contract: run Hardhat locally or deploy from your machine with PRIVATE_KEY and Alchemy/Infura.

## Security & Next steps

Important notes / next steps I recommend (practical)
