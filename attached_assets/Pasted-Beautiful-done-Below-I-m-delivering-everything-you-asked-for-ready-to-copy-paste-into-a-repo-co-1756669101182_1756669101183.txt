pragma solidity ^0.8.17;

interface IReceipt {
    /// @notice Mint an immutable receipt NFT representing a ceremony/match/transaction

    /// @param uri IPFS metadata URI
    /// @param ceremonyHash Keccak256 hash of the serialized ceremony/match record

    function mintReceipt(address to, string calldata uri, bytes32 ceremonyHash) external returns (uint256);

    event ReceiptMinted(uint256 indexed tokenId, address indexed to, string uri, bytes32 ceremonyHash);
}

2) contracts/GuardianManager.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "@openzeppelin/contracts/token/ERC721/extensions/ERC721Enumerable.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

interface IReceipt {
    function mintReceipt(address to, string calldata uri, bytes32 ceremonyHash) external returns (uint256);
}

contract GuardianManager is ERC721Enumerable, Ownable {
    IReceipt public receiptContract;
    uint256 public nextTokenId = 1000;

    // tier supply caps
    uint256 public constant MAX_OMNI = 10;
    uint256 public constant MAX_GOLD = 1000;
    // counters
    uint256 public omniMinted;
    uint256 public goldMinted;

    mapping(uint256 => bytes32) public ceremonyHashOf;
    mapping(uint256 => uint8) public tierOf; // 0: bronze,1:silver,2:gold,3:omni

    event GuardianMinted(uint256 indexed tokenId, address indexed to, uint8 tier, bytes32 ceremonyHash);

    constructor(address _receipt) ERC721("GuardianManager", "GUARD") {
        receiptContract = IReceipt(_receipt);
        nextTokenId = 1000; // reserve ranges
    }

    function setReceipt(address _r) external onlyOwner {
        receiptContract = IReceipt(_r);
    }

    function _mintGuardian(address to, string memory tokenURI, bytes32 ceremonyHash, uint8 tier) internal returns (uint256) {
        if (tier == 3) { // Omni
            require(omniMinted < MAX_OMNI, "Omni supply exhausted");
            omniMinted += 1;
        } else if (tier == 2) {
            require(goldMinted < MAX_GOLD, "Gold supply exhausted");
            goldMinted += 1;
        }
        uint256 tid = nextTokenId++;
        _safeMint(to, tid);
        // store ceremony
        ceremonyHashOf[tid] = ceremonyHash;
        tierOf[tid] = tier;
        emit GuardianMinted(tid, to, tier, ceremonyHash);

        // optionally, call external receipt minter for cross-proof (if set)
        if (address(receiptContract) != address(0)) {
            // we pass a minimal receipt with the same metadata URI so provenance is recorded there too
            try receiptContract.mintReceipt(to, tokenURI, ceremonyHash) returns (uint256) {
                // receipt minted
            } catch {
                // do not revert - allow recording on-chain separately
            }
        }

        return tid;
    }

    /// @notice Public admin mint (used by backend after verification)
    function adminMint(address to, string calldata tokenURI, bytes32 ceremonyHash, uint8 tier) external onlyOwner returns (uint256) {
        return _mintGuardian(to, tokenURI, ceremonyHash, tier);
    }

    /// @notice helper to burn tokens (rare)
    function adminBurn(uint256 tokenId) external onlyOwner {
        _burn(tokenId);
    }

    /// @notice Override tokenURI if you want on-chain pointer; using baseURI behavior
    string private _baseTokenURI;
    function _baseURI() internal view override returns (string memory) {
        return _baseTokenURI;
    }
    function setBaseURI(string calldata u) external onlyOwner {
        _baseTokenURI = u;
    }
}

6) pages/api/ignite.js (Next.js serverless function)

This endpoint expects a signed, validated match object and mints the Guardian NFT by:
	•	compute ceremony_hash
	•	store match in DB (pseudo)
	•	pin metadata & match record to IPFS (using Pinata/Estuary)
	•	call on-chain mint via IReceipt or GuardianManager admin route

This is a production skeleton — wire your DB and pinning providers (keys via env).

// pages/api/ignite.js
import { ethers } from "ethers";
import fetch from "node-fetch";
import pinataSDK from "@pinata/sdk"; // or use your provider
const PINATA_KEY = process.env.PINATA_KEY;
const PINATA_SECRET = process.env.PINATA_SECRET;
const pinata = PINATA_KEY ? pinataSDK(PINATA_KEY, PINATA_SECRET) : null;

const IRECEIPT_ADDRESS = process.env.IRECEIPT_ADDRESS;
const IRECEIPT_ABI = require('../../../abi/IReceipt.json');

export default async function handler(req, res) {
  if (req.method !== "POST") return res.status(405).end();
  try {
    const payload = req.body; // matchID, winner, loser, roundScores, metadata, winnerAddress
    // 1. compute ceremony_hash
    const serialized = JSON.stringify(payload);
    const ceremonyHash = ethers.utils.keccak256(ethers.utils.toUtf8Bytes(serialized));

    // 2. Save to DB (pseudo - replace with Supabase/Postgres logic)
    // await supabase.from('matches').insert([{...payload, ceremonyHash}]);

    // 3. Pin match record to IPFS
    if (!pinata) throw new Error("Pinata not configured in env");
    const matchPin = await pinata.pinJSONToIPFS({ ...payload, ceremonyHash });
    const matchIpfs = `ipfs://${matchPin.IpfsHash}`;

    // 4. Build NFT metadata and pin
    const nftMetadata = {
      name: payload.metadata?.name || `Guardian - ${payload.matchID}`,
      description: payload.metadata?.description || "Guardian Challenge receipt",
      image: payload.metadata?.image || payload.metadata?.imagePlaceholder,
      attributes: payload.metadata?.attributes || [],
      ceremony_hash: ceremonyHash,
      matchID: payload.matchID,
      provenanceURI: matchIpfs
    };
    const metaPin = await pinata.pinJSONToIPFS(nftMetadata);
    const tokenUri = `ipfs://${metaPin.IpfsHash}`;

    // 5. Call contract with server-side signer
    const provider = new ethers.providers.JsonRpcProvider(process.env.RPC_URL);
    const signer = new ethers.Wallet(process.env.DEPLOYER_PRIVATE_KEY, provider);
    const receipt = new ethers.Contract(IRECEIPT_ADDRESS, IRECEIPT_ABI, signer);
    const tx = await receipt.mintReceipt(payload.winnerAddress, tokenUri, ceremonyHash);
    const txr = await tx.wait();

    // return result
    res.status(200).json({ ok: true, txHash: txr.transactionHash, tokenUri, matchIpfs, ceremonyHash });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: e.message });
  }
}

Security note: DEPLOYER_PRIVATE_KEY must be stored in GitHub/Vercel secrets. Use a multisig for mainnet.

7) components/MintButton.js (React, wagmi)

import React from 'react';
import { useSigner } from 'wagmi';

export default function MintButton({ matchId, metadata }) {
  const { data: signer } = useSigner();
  const mint = async () => {
    const res = await fetch('/api/ignite', {
      method: 'POST',
      headers: {'Content-Type':'application/json'},
      body: JSON.stringify({
        matchID: matchId,
        winnerAddress: await signer.getAddress(),
        metadata
      })
    });
    const j = await res.json();
    if (j.ok) {
      alert('Minted! tx: ' + j.txHash);
    } else {
      alert('Error: ' + (j.error || 'Unknown'));
    }
  };
  return <button onClick={mint} className="mint-button">Finalize & Mint Guardian NFT</button>
}
